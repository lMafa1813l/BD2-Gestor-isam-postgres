CREATE TABLE product_inventory (
    product_partNumber          VARCHAR(50) PRIMARY KEY, 
    location_locationIdentifier VARCHAR(50) NOT NULL,
    inventoryType               VARCHAR(30) NOT NULL,
    quantity                    INT,
    quantityUnits               VARCHAR(10) NOT NULL,
    value                       DECIMAL(15, 2) NOT NULL,
    valueCurrency               VARCHAR(10) NOT NULL,
    reservationOrders           INT,
    daysOfSupply                INT,
    shelfLife                   INT,
    reorderLevel                INT,
    expectedLeadTime            INT,
    quantityUpperThreshold      INT,
    quantityLowerThreshold      INT,
    daysOfSupplyUpperThreshold  INT,
    daysOfSupplyLowerThreshold  INT,
    expiringThreshold           INT,
    plannerCode                 VARCHAR(50),
    velocityCode                VARCHAR(5),
    inventoryParentType         VARCHAR(30),
    class                       VARCHAR(20),
    segment                     VARCHAR(30)
);
